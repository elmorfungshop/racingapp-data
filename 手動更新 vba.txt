Sub ExportDataToJson()
    Dim ws As Worksheet
    Dim jsonFile As String
    Dim jsonStr As String
    Dim firstSheet As Boolean
    Dim firstRow As Boolean
    Dim i As Long, j As Long
    Dim lastRow As Long, lastCol As Long
    Dim cellVal As String
    Dim headers() As String
    Dim headerRowIdx As Long
    Dim rStr As String
    Dim stream As Object
    Dim hasData As Boolean
    Dim rowStrCheck As String
    Dim firstHeader As Boolean
    Dim firstCell As Boolean

    ' 設定輸出的 JSON 檔案路徑（與 Excel 檔案同資料夾）
    jsonFile = ThisWorkbook.Path & "\data.json"

    jsonStr = "{" & vbCrLf
    firstSheet = True

    ' 迴圈走訪活頁簿中的每一個工作表（每一場賽事）
    For Each ws In ThisWorkbook.Worksheets
        ' 略過隱藏的工作表
        If ws.Visible = xlSheetVisible Then
            
            If Not firstSheet Then
                jsonStr = jsonStr & "," & vbCrLf
            End If
            firstSheet = False
            
            ' 1. 輸出工作表名稱作為 Key
            jsonStr = jsonStr & "  """ & EscapeJson(ws.Name) & """: {" & vbCrLf
            
            lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
            
            ' 自動尋找標題列位置
            headerRowIdx = 1
            For i = 1 To Application.Min(lastRow, 5)
                rStr = ""
                For j = 1 To 10
                    rStr = rStr & " " & ws.Cells(i, j).Value
                Next j
                If InStr(rStr, "馬號") > 0 Or InStr(rStr, "馬名") > 0 Or InStr(rStr, "檔位") > 0 Then
                    headerRowIdx = i
                    Exit For
                End If
            Next i
            
            ' 取得標題行數值
            lastCol = ws.Cells(headerRowIdx, ws.Columns.Count).End(xlToLeft).Column
            ReDim headers(1 To lastCol)
            
            ' 2. 輸出 headers 陣列（自動過濾「綵衣」）
            jsonStr = jsonStr & "    ""headers"": ["
            firstHeader = True
            For j = 1 To lastCol
                headers(j) = Trim(CStr(ws.Cells(headerRowIdx, j).Value))
                If headers(j) = "" Then headers(j) = "Col" & j
                
                ' 如果標題不是「綵衣」，才輸出
                If InStr(headers(j), "綵衣") = 0 Then
                    If Not firstHeader Then jsonStr = jsonStr & ", "
                    jsonStr = jsonStr & """" & EscapeJson(headers(j)) & """"
                    firstHeader = False
                End If
            Next j
            jsonStr = jsonStr & "]," & vbCrLf
            
            ' 3. 輸出 rows 陣列（自動過濾「綵衣」對應的資料格）
            jsonStr = jsonStr & "    ""rows"": [" & vbCrLf
            firstRow = True
            
            For i = headerRowIdx + 1 To lastRow
                hasData = False
                For j = 1 To lastCol
                    If ws.Cells(i, j).Value <> "" Then
                        hasData = True
                        Exit For
                    End If
                Next j
                
                If hasData Then
                    rowStrCheck = ""
                    For j = 1 To lastCol
                        rowStrCheck = rowStrCheck & " " & ws.Cells(i, j).Value
                    Next j
                    
                    ' 過濾含有「全餐」或「(退出)」的無效行
                    If InStr(rowStrCheck, "全餐") = 0 And InStr(rowStrCheck, "(退出)") = 0 Then
                        
                        If Not firstRow Then
                            jsonStr = jsonStr & "," & vbCrLf
                        End If
                        firstRow = False
                        
                        jsonStr = jsonStr & "      ["
                        firstCell = True
                        For j = 1 To lastCol
                            ' 如果原標題不是「綵衣」，才輸出該欄數值
                            If InStr(headers(j), "綵衣") = 0 Then
                                If Not firstCell Then jsonStr = jsonStr & ", "
                                cellVal = CStr(ws.Cells(i, j).Value)
                                
                                ' 數字保持數值格式，文字加上引號
                                If IsNumeric(cellVal) And cellVal <> "" Then
                                    jsonStr = jsonStr & cellVal
                                Else
                                    jsonStr = jsonStr & """" & EscapeJson(cellVal) & """"
                                End If
                                firstCell = False
                            End If
                        Next j
                        jsonStr = jsonStr & "]"
                    End If
                End If
            Next i
            
            jsonStr = jsonStr & vbCrLf & "    ]" & vbCrLf & "  }"
        End If
    Next ws

    jsonStr = jsonStr & vbCrLf & "}"

    ' 使用 ADODB.Stream 以 UTF-8 格式寫入檔案
    Set stream = CreateObject("ADODB.Stream")
    With stream
        .Type = 2
        .Charset = "utf-8"
        .Open
        .WriteText jsonStr
        .SaveToFile jsonFile, 2
        .Close
    End With

    MsgBox "成功產生 data.json（已自動排除綵衣欄位，恢復原網頁對應）！", vbInformation, "完成"
End Sub

' 輔助函數：正確跳脫 JSON 特殊字元
Function EscapeJson(str As String) As String
    Dim result As String
    result = str
    result = Replace(result, "\", "\\")
    result = Replace(result, """", "\""")
    result = Replace(result, vbCrLf, " ")
    result = Replace(result, vbCr, " ")
    result = Replace(result, vbLf, " ")
    EscapeJson = result
End Function


Sub RefreshAndExport()
    On Error GoTo ErrorHandler
    
    ' 1. 自動重新整理 Excel 中所有的外部資料連線 (Power Query)
    ThisWorkbook.RefreshAll
    
    ' 【重要提示】
    ' 因為 RefreshAll 是非同步執行，建議您在 Excel 的「資料 > 查詢與連線」中，
    ' 將相關 Power Query 連線的內容屬性裡面的「啟用背景重新整理 (Enable background refresh)」取消勾選。
    ' 這樣 VBA 才能真正等待資料抓完才往下走。這裡暫停 2 秒作為額外緩衝。
    Application.Wait (Now + TimeValue("0:00:02"))
    
    ' 2. 執行剛才的轉 JSON 程式 (名稱已對應為 ExportDataToJson)
    Call ExportDataToJson
    
    Exit Sub

ErrorHandler:
    MsgBox "發生錯誤：" & Err.Description, vbCritical, "錯誤"
End Sub
